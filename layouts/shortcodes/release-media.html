{{- /* A screenshot or screen recording in a release post. Usage: {{< release-media id="pr1234-1" alt="What it shows" >}} The files are in a GitHub release (release-post/assets.mjs). media.json has their URLs and sizes, in the item with this id. */ -}} {{- $id := .Get "id" -}} {{- $alt := .Get "alt" -}} {{- $item := dict -}} {{- with .Page.Resources.Get "media.json" -}} {{- range (transform.Unmarshal .).items -}} {{- if eq .id $id -}}{{- $item = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $main := index ($item.files | default dict) "main" -}} {{- if not $main -}} {{- errorf "release-media: %q is not uploaded (no files in media.json) in %s" $id .Page.File.Path -}} {{- end -}} {{- $url := `^https://github\.com/[\w.-]+/[\w.-]+/releases/download/blog-media-v[\d.]+/[\w.-]+$` -}} {{- if not (findRE $url $main.url) -}} {{- errorf "release-media: bad URL for %q in %s" $id .Page.File.Path -}} {{- end -}} {{- /* The width and height attributes give the browser the shape before the file loads. The width is at most the column, and small enough that a tall phone picture stays under 75% of the screen height. */ -}} {{- $w := int $main.width -}} {{- $h := int $main.height -}} {{- $style := printf "width: min(100%%, calc(75vh * %d / %d)); height: auto;" $w $h -}}
