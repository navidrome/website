{{- /* Audio player for a release post. The MP3 is in a GitHub release (release-post/assets.mjs), and audio.json has its URL. Renders nothing until then. GitHub serves release files as application/octet-stream. Safari plays them only when the source says its type, so the URL goes in a source tag. */ -}} {{- with .Page.Resources.Get "audio.json" -}} {{- with (transform.Unmarshal .).url -}} {{- if not (findRE `^https://github\.com/[\w.-]+/[\w.-]+/releases/download/blog-media-v[\d.]+/[\w.-]+$` .) -}} {{- errorf "release-audio: bad audio URL %q in %s" . $.Page.File.Path -}} {{- end -}} {{- end -}} {{- end -}}
